# Baseline, executed serially in the current-worktree Docker container:
make rv64-kernel-build-only
make la64-kernel-build-only

# Make parse-time contract:
make -f make/rv64.mk -n build CORE_NUM={1,2,4,8}
make -f make/rv64.mk -n build CORE_NUM=3
make -f make/la64.mk -n build CORE_NUM={1,2,4,8}
make -f make/la64.mk -n build CORE_NUM=3

# Actual top-level entry points, both expected to reject before Cargo:
make rv64-kernel-build-only CORE_NUM=3
make la64-kernel-build-only CORE_NUM=3

# Export contract, using a temporary in-memory Make target:
make -s -f make/rv64.mk --eval='print-smp-core: ; @env | grep "^MANGO_CORE_NUM="' print-smp-core CORE_NUM=2
make -s -f make/la64.mk --eval='print-smp-core: ; @env | grep "^MANGO_CORE_NUM="' print-smp-core CORE_NUM=2

# Direct build.rs contract:
rustc build.rs -o <temporary-directory>/build-script-check
MANGO_CORE_NUM=2 <temporary-directory>/build-script-check
MANGO_CORE_NUM=3 <temporary-directory>/build-script-check

# Post-change builds, executed serially:
make rv64-kernel-build-only CORE_NUM=1
make la64-kernel-build-only CORE_NUM=1

# Post-change single-core focused QEMU tests, executed serially:
make rv64-ktest KTEST=waitqueue KREPEAT=1 CORE_NUM=1 KTEST_QEMU_TIMEOUT=45
make la64-ktest KTEST=waitqueue KREPEAT=1 CORE_NUM=1 KTEST_QEMU_TIMEOUT=45
